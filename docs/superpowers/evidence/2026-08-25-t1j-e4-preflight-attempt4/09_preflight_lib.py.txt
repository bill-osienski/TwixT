"""E4 preflight — the decision logic, and the controls that prove it rejects.

Every threshold comes from the PREREGISTRATION FILE, never from a literal here;
`load_prereg` refuses to run if the file's hash does not match the one frozen
before measurement began.

NO GAMES. Nothing in this module plays, scores, or consumes a seed.
"""

def load_prereg(path, expected_sha256):
    raw = open(path, "rb").read()
    got = hashlib.sha256(raw).hexdigest()
    if got != expected_sha256:
        raise SystemExit(f"PREREGISTRATION HASH MISMATCH: {got} != {expected_sha256}")
    return json.loads(raw)


# ---------------------------------------------------------------- per query

def check_query(rec, *, depth, our_state, searched, bound, transform=A.CANONICAL):
    """Divergences for one query. Empty list == acceptable."""
    d = []
    if rec.requested_depth != depth:
        d.append(f"requested_depth {rec.requested_depth} != {depth}")
    if not rec.usealphabeta:
        d.append("usealphabeta false — no search ran")
    if rec.current_max_ply != depth + 1:
        d.append(f"currentMaxPly {rec.current_max_ply} != {depth + 1} — depth not completed")
    if not rec.completed:
        d.append("helper reported completed=false")
    if rec.null_sentinel or rec.move is None:
        d.append("returned the null sentinel (-1,-1)")
    if not rec.legal:
        d.append("T1j reports its own move illegal")
    if rec.to_move != A.PLAYER_TO_T1J[our_state.to_move]:
        d.append(f"side to move {rec.to_move} != {A.PLAYER_TO_T1J[our_state.to_move]}")
    if rec.move is not None and rec.move not in set(our_state.legal_moves()):
        d.append(f"move {rec.move} is not legal in OUR engine")
    if rec.move_nr != our_state.ply:
        d.append(f"moveNr {rec.move_nr} != our ply {our_state.ply}")
    # the query process must have searched the position we bound through E3b
    for field in ("pegs", "bridges", "next_player", "ply", "legal", "term_y", "term_x"):
        a, b = getattr(searched, field), getattr(bound, field)
        if a != b:
            d.append(f"searched position differs from the bound position on {field}")
    return d


# ------------------------------------------------------------ dial response

def classify_dial(by_depth_moves, completed_all):
    """by_depth_moves: {position: {depth: move}}. Returns (label, detail).

    Identical moves are NOT proof the dial is dead -- a deeper search can
    correctly re-select the same move -- so that case is INCONCLUSIVE, not a
    negative result.
    """
    if not completed_all:
        return "dial_failure", "at least one requested depth did not complete"
    changed = []
    for pos, by_d in sorted(by_depth_moves.items()):
        if 3 not in by_d:
            return "dial_failure", f"{pos}: depth 3 baseline missing"
        base = by_d[3]
        for d in sorted(by_d):
            if d != 3 and by_d[d] != base:
                changed.append(f"{pos}: depth {d} -> {by_d[d]} vs depth 3 -> {base}")
    if changed:
        return "observable_move_response", changed
    return "no_observed_move_response_inconclusive", (
        "every move identical across all measured depths, and every requested "
        "depth demonstrably completed")


# ------------------------------------------------- strong-endpoint selection

def select_strong_endpoint(stats, prereg, measured_depths):
    """stats: {depth: {'max_wall_ms': int, 'max_engine_ms': int, 'all_completed', 'n'}}.

    COST ONLY, and specifically WALL ms -- the metric the preregistration names.
    Attempt 1 fed this the helper's INTERNAL engine time, so the gate did not
    compute the statistic it had frozen. `max_engine_ms` is carried alongside for
    the record and is deliberately never read here.

    The moves returned are not an input, and are not passed in.
    """
    assert prereg["cost_metric"]["selector_metric"] == "wall_ms", \
        "this selector implements the wall-ms rule only"
    rule = prereg["strong_endpoint_selection_rule"]
    max_ms = rule["max_query_ms"]
    det_budget_ms = prereg["budgets"]["determinism_budget_s"] * 1000
    timeout_ms = prereg["budgets"]["per_query_timeout_s"] * 1000
    reasons, ok = [], []
    for d in sorted(measured_depths):
        s = stats.get(d)
        if s is None:
            reasons.append(f"depth {d}: not measured"); continue
        conds = [
            (s["all_completed"], "every position completed"),
            (s["max_wall_ms"] <= timeout_ms,
             f"max wall {s['max_wall_ms']}ms <= timeout {timeout_ms}ms"),
            (s["max_wall_ms"] <= max_ms,
             f"max wall {s['max_wall_ms']}ms <= max_query_ms {max_ms}ms"),
            (25 * s["max_wall_ms"] <= det_budget_ms,
             f"projected determinism 25*{s['max_wall_ms']}ms wall <= {det_budget_ms}ms"),
        ]
        passed = all(c for c, _ in conds)
        reasons.append(f"depth {d}: {'QUALIFIES' if passed else 'rejected'} — "
                       + "; ".join(f"{'ok' if c else 'NO'} {t}" for c, t in conds))
        if passed:
            ok.append(d)
    if not ok:
        return None, reasons
    return max(ok), reasons


# ------------------------------------------------- endpoint-screen decisions
# FROZEN RULES. Nothing here plays a game; these classify a result vector that
# only a later, separately authorized screen could produce.

DECISIONS = ("SATURATED_STRONG", "SATURATED_WEAK", "IN_BAND", "INCOMPLETE")


def per_endpoint_decision(score, played, n, band, cap_terminations=0):
    """T1j's decision at ONE endpoint. `score` counts draws as 0.5.

    A draw is always a ply-cap termination -- TwixT cannot draw by rule -- so an
    endpoint where most games hit the cap is INCOMPLETE, not scored.
    """
    lo, hi = band[0] * n, band[1] * n
    if cap_terminations * 2 > n:
        return "INCOMPLETE"
    if played < n:
        # only a FORCED IN_BAND may stop early -- and forcing it takes more than
        # ruling out score saturation; see early_in_band_forced
        if early_in_band_forced(score, played, n, band, cap_terminations):
            return "IN_BAND"
        return "INCOMPLETE"
    if score >= hi:
        return "SATURATED_STRONG"
    if score <= lo:
        return "SATURATED_WEAK"
    return "IN_BAND"


def saturation_reachable(score, played, n, band):
    """Can EITHER SCORE saturation verdict still be reached from here?"""
    lo, hi = band[0] * n, band[1] * n
    strong_possible = score + (n - played) >= hi
    weak_possible = score <= lo
    return strong_possible or weak_possible


def cap_incompleteness_reachable(played, n, cap_terminations):
    """Could this endpoint still end INCOMPLETE through ply-cap terminations?

    The cap-saturation abort fires at cap_terminations * 2 > n. Every unplayed
    game could still hit the cap, so the abort remains reachable while
    cap_terminations + (n - played) > n // 2.
    """
    return cap_terminations + (n - played) > n // 2


def early_in_band_forced(score, played, n, band, cap_terminations=0):
    """Is IN_BAND the ONLY outcome still reachable?

    Attempt 2 checked score saturation alone and concluded IN_BAND was forced
    after one win and one loss. It is NOT: with 14 games unplayed, nine of them
    could still terminate at the ply cap, firing the cap-saturation abort and
    making the endpoint INCOMPLETE. Both routes out of IN_BAND must be closed.
    """
    if played >= n:
        return False                       # not an EARLY stop
    return (not saturation_reachable(score, played, n, band)
            and not cap_incompleteness_reachable(played, n, cap_terminations))


def earliest_early_stop(n, band):
    """The first game at which an IN_BAND early stop can possibly be forced.

    Reported rather than asserted: the number follows from n and the band, and
    stating it wrongly is exactly the defect this function exists to prevent.
    """
    for played in range(1, n + 1):
        for half in range(0, 2 * played + 1):
            if early_in_band_forced(half / 2.0, played, n, band, 0):
                return played, half / 2.0
    return None, None


JOINT = {
    ("SATURATED_STRONG", "SATURATED_STRONG"): "T1J_TOO_STRONG",
    ("SATURATED_WEAK", "SATURATED_WEAK"): "T1J_TOO_WEAK",
    ("SATURATED_WEAK", "SATURATED_STRONG"): "BRACKETED",
    ("SATURATED_STRONG", "SATURATED_WEAK"): "BRACKETED",
}


def classify_joint(decision_weak, decision_strong):
    """TOTAL over all 16 combinations of DECISIONS; asserts its own totality."""
    for d in (decision_weak, decision_strong):
        if d not in DECISIONS:
            raise ValueError(f"unknown endpoint decision {d!r}")
    if "INCOMPLETE" in (decision_weak, decision_strong):
        return "INCONCLUSIVE"
    if "IN_BAND" in (decision_weak, decision_strong):
        return "IN_BAND"
    return JOINT[(decision_weak, decision_strong)]


LARGER_MATCH_PERMITTED = ("IN_BAND", "BRACKETED", "INCONCLUSIVE")


def joint_truth_table():
    """Every combination, so the frozen plan carries the whole mapping."""
    return {f"{a}|{b}": classify_joint(a, b) for a in DECISIONS for b in DECISIONS}


# ------------------------------------------------------------------ selftest

class _Rec:
    def __init__(self, **kw):
        base = dict(q=1, requested_depth=5, move=(15, 15), to_move="Y", usealphabeta=True,
                    current_max_ply=6, completed_depth=5, completed=True, legal=True,
                    null_sentinel=False, move_nr=6, eval_regime="normal", elapsed_us=10)
        base.update(kw)
        self.__dict__.update(base)


class _State:
    to_move = "red"
    ply = 6
    def legal_moves(self): return [(15, 15), (1, 1)]


class _Dump:
    def __init__(self, **kw):
        base = dict(pegs={"a"}, bridges=set(), next_player="Y", ply=6,
                    legal={(0, 0)}, term_y=False, term_x=False)
        base.update(kw)
        self.__dict__.update(base)


def selftest():
    bad = 0

    def case(name, got_bad, want_bad):
        nonlocal bad
        ok = bool(got_bad) == want_bad
        print(f"  {'PASS' if ok else 'FAIL'}  {name}"
              + (f"   [{got_bad[0][:76]}]" if got_bad else ""))
        if not ok: bad += 1

    st, dump = _State(), _Dump()
    print("preflight comparator self-test (negative controls)")
    case("baseline accepted", check_query(_Rec(), depth=5, our_state=st, searched=dump, bound=dump), False)
    case("wrong requested depth rejected",
         check_query(_Rec(requested_depth=4), depth=5, our_state=st, searched=dump, bound=dump), True)
    case("usealphabeta false rejected",
         check_query(_Rec(usealphabeta=False), depth=5, our_state=st, searched=dump, bound=dump), True)
    case("currentMaxPly != depth+1 rejected (depth NOT completed)",
         check_query(_Rec(current_max_ply=4), depth=5, our_state=st, searched=dump, bound=dump), True)
    case("completed=false rejected",
         check_query(_Rec(completed=False), depth=5, our_state=st, searched=dump, bound=dump), True)
    case("null sentinel rejected",
         check_query(_Rec(move=None, null_sentinel=True), depth=5, our_state=st, searched=dump, bound=dump), True)
    case("T1j-illegal move rejected",
         check_query(_Rec(legal=False), depth=5, our_state=st, searched=dump, bound=dump), True)
    case("move illegal in OUR engine rejected",
         check_query(_Rec(move=(23, 23)), depth=5, our_state=st, searched=dump, bound=dump), True)
    case("side-to-move divergence rejected",
         check_query(_Rec(to_move="X"), depth=5, our_state=st, searched=dump, bound=dump), True)
    case("ply divergence rejected",
         check_query(_Rec(move_nr=7), depth=5, our_state=st, searched=dump, bound=dump), True)
    case("searched != bound position rejected",
         check_query(_Rec(), depth=5, our_state=st, searched=_Dump(pegs={"b"}), bound=dump), True)

    print("dial classification")
    for name, moves, comp, want in (
        ("all identical + completed -> INCONCLUSIVE, not 'dead'",
         {"p": {3: (1, 1), 5: (1, 1), 7: (1, 1)}}, True, "no_observed_move_response_inconclusive"),
        ("one changed move -> observable response",
         {"p": {3: (1, 1), 5: (2, 2)}}, True, "observable_move_response"),
        ("a depth did not complete -> dial failure",
         {"p": {3: (1, 1), 5: (1, 1)}}, False, "dial_failure"),
    ):
        got, _ = classify_dial(moves, comp)
        ok = got == want
        print(f"  {'PASS' if ok else 'FAIL'}  {name}: {got}")
        if not ok: bad += 1

    print("strong-endpoint selection (cost only)")
    prereg = {"strong_endpoint_selection_rule": {"max_query_ms": 30000},
              "budgets": {"determinism_budget_s": 900, "per_query_timeout_s": 120},
              "cost_metric": {"selector_metric": "wall_ms"}}
    checks = [
        ("deepest qualifying depth chosen",
         {3: dict(max_wall_ms=10, max_engine_ms=5, all_completed=True, n=6), 5: dict(max_wall_ms=900, max_engine_ms=450, all_completed=True, n=6)},
         [3, 5], 5),
        ("a depth that did not complete is rejected",
         {3: dict(max_wall_ms=10, max_engine_ms=5, all_completed=True, n=6), 5: dict(max_wall_ms=10, max_engine_ms=5, all_completed=False, n=6)},
         [3, 5], 3),
        ("over max_query_ms rejected",
         {3: dict(max_wall_ms=10, max_engine_ms=5, all_completed=True, n=6), 5: dict(max_wall_ms=30001, max_engine_ms=5, all_completed=True, n=6)},
         [3, 5], 3),
        ("determinism projection over budget rejected (25 * 36001ms > 900s)",
         {3: dict(max_wall_ms=10, max_engine_ms=5, all_completed=True, n=6), 5: dict(max_wall_ms=36001, max_engine_ms=5, all_completed=True, n=6)},
         [3, 5], 3),
        ("nothing qualifies -> None",
         {3: dict(max_wall_ms=99999999, max_engine_ms=1, all_completed=True, n=6)}, [3], None),
    ]
    for name, stats, measured, want in checks:
        got, _ = select_strong_endpoint(stats, prereg, measured)
        ok = got == want
        print(f"  {'PASS' if ok else 'FAIL'}  {name}: chose {got}")
        if not ok: bad += 1

    print("endpoint-screen decision rules (frozen; nothing is played)")
    N, BAND = 16, [0.05, 0.95]
    for name, args, want in (
        ("clean sweep for T1j -> SATURATED_STRONG", (16.0, 16, N, BAND), "SATURATED_STRONG"),
        ("15.5/16 -> SATURATED_STRONG (>= 15.2)", (15.5, 16, N, BAND), "SATURATED_STRONG"),
        ("15.0/16 -> IN_BAND (below 15.2)", (15.0, 16, N, BAND), "IN_BAND"),
        ("0/16 -> SATURATED_WEAK", (0.0, 16, N, BAND), "SATURATED_WEAK"),
        ("0.5/16 -> SATURATED_WEAK (<= 0.8)", (0.5, 16, N, BAND), "SATURATED_WEAK"),
        ("1.0/16 -> IN_BAND (above 0.8)", (1.0, 16, N, BAND), "IN_BAND"),
        ("8/16 all draws -> IN_BAND", (8.0, 16, N, BAND), "IN_BAND"),
        ("unfinished and saturation still reachable -> INCOMPLETE", (0.0, 4, N, BAND), "INCOMPLETE"),
        ("1 win 1 loss after 2 -> still INCOMPLETE (cap route open)", (1.0, 2, N, BAND), "INCOMPLETE"),
        ("4-4 after 8 with no caps -> IN_BAND, EARLY", (4.0, 8, N, BAND), "IN_BAND"),
    ):
        got = per_endpoint_decision(*args)
        ok = got == want
        print(f"  {'PASS' if ok else 'FAIL'}  {name}: {got}")
        if not ok: bad += 1
    got = per_endpoint_decision(8.0, 16, N, BAND, cap_terminations=9)
    ok = got == "INCOMPLETE"
    print(f"  {'PASS' if ok else 'FAIL'}  9 of 16 games hit the ply cap -> {got}")
    if not ok: bad += 1

    ok = not saturation_reachable(1.0, 2, N, BAND)
    print(f"  {'PASS' if ok else 'FAIL'}  score saturation impossible after 1 win + 1 loss")
    if not ok: bad += 1
    ok = cap_incompleteness_reachable(2, N, 0)
    print(f"  {'PASS' if ok else 'FAIL'}  but CAP incompleteness is STILL reachable at game 2 "
          f"(0 caps + 14 unplayed > 8) — attempt 2 missed this")
    if not ok: bad += 1
    ok = not early_in_band_forced(1.0, 2, N, BAND, 0)
    print(f"  {'PASS' if ok else 'FAIL'}  so IN_BAND is NOT forced at game 2")
    if not ok: bad += 1
    ok = early_in_band_forced(4.0, 8, N, BAND, 0)
    print(f"  {'PASS' if ok else 'FAIL'}  IN_BAND IS forced at game 8 with 0 caps and 4-4")
    if not ok: bad += 1
    ok = not early_in_band_forced(4.0, 8, N, BAND, 1)
    print(f"  {'PASS' if ok else 'FAIL'}  NEGATIVE CONTROL: one cap termination at game 8 "
          f"reopens the cap route, so the early stop is refused")
    if not ok: bad += 1
    ok = not early_in_band_forced(7.5, 8, N, BAND, 0)
    print(f"  {'PASS' if ok else 'FAIL'}  NEGATIVE CONTROL: 7.5 of 8 leaves strong saturation "
          f"reachable (7.5+8 >= 15.2), so the early stop is refused")
    if not ok: bad += 1
    ok = not early_in_band_forced(8.0, 16, N, BAND, 0)
    print(f"  {'PASS' if ok else 'FAIL'}  a completed endpoint is not an EARLY stop")
    if not ok: bad += 1
    played, sc = earliest_early_stop(N, BAND)
    ok = played == 8
    print(f"  {'PASS' if ok else 'FAIL'}  earliest possible early stop is game {played} "
          f"(score {sc}), not game 2 as attempt 2 claimed")
    if not ok: bad += 1
    ok = saturation_reachable(0.0, 15, N, BAND)
    print(f"  {'PASS' if ok else 'FAIL'}  a SATURATED verdict is NOT reachable before game 16 "
          f"(still reachable at 15)")
    if not ok: bad += 1

    table = joint_truth_table()
    ok = len(table) == len(DECISIONS) ** 2 and all(v for v in table.values())
    print(f"  {'PASS' if ok else 'FAIL'}  joint classifier is TOTAL over all "
          f"{len(DECISIONS)}^2 = {len(table)} combinations")
    if not ok: bad += 1
    for a, b, want in (
        ("IN_BAND", "IN_BAND", "IN_BAND"),
        ("SATURATED_WEAK", "SATURATED_STRONG", "BRACKETED"),
        ("SATURATED_STRONG", "SATURATED_STRONG", "T1J_TOO_STRONG"),
        ("SATURATED_WEAK", "SATURATED_WEAK", "T1J_TOO_WEAK"),
        ("INCOMPLETE", "IN_BAND", "INCONCLUSIVE"),
        ("SATURATED_STRONG", "IN_BAND", "IN_BAND"),
    ):
        got = classify_joint(a, b)
        ok = got == want
        print(f"  {'PASS' if ok else 'FAIL'}  joint({a}, {b}) = {got}")
        if not ok: bad += 1
    try:
        classify_joint("NONSENSE", "IN_BAND"); print("  FAIL  unknown decision accepted"); bad += 1
    except ValueError:
        print("  PASS  unknown endpoint decision rejected")

    # the exact defect from attempt 1: a depth whose ENGINE time is tiny but whose
    # WALL time is over the ceiling must be rejected. Reading engine ms would pass it.
    got, _ = select_strong_endpoint(
        {3: dict(max_wall_ms=10, max_engine_ms=5, all_completed=True, n=6),
         5: dict(max_wall_ms=30001, max_engine_ms=12, all_completed=True, n=6)},
        prereg, [3, 5])
    ok = got == 3
    print(f"  {'PASS' if ok else 'FAIL'}  rejects on WALL ms even when engine ms is tiny: chose {got}")
    if not ok: bad += 1

    # the rule must be blind to moves: it takes no move argument at all
    import inspect
    params = set(inspect.signature(select_strong_endpoint).parameters)
    ok = params == {"stats", "prereg", "measured_depths"}
    print(f"  {'PASS' if ok else 'FAIL'}  selection rule takes no move input (params {sorted(params)})")
    if not ok: bad += 1

    print("SELFTEST " + ("OK" if not bad else f"FAILED ({bad})"))
    return bad == 0


if __name__ == "__main__":
    sys.exit(0 if selftest() else 2)
