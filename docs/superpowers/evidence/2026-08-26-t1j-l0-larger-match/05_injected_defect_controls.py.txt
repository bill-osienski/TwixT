def h(p): return hashlib.sha256(open(p, "rb").read()).hexdigest()


def purge_bytecode():
    """CPython invalidates a .pyc on (source mtime SECONDS, source size).

    A same-length edit inside one second therefore leaves the stale .pyc valid
    and the injected defect never reaches the interpreter -- the control then
    passes vacuously. `N_REPS = 4` -> `N_REPS = 3` did exactly that. Purge, and
    forbid writing new ones, so every run compiles the bytes on disk.
    """
    for root, dirs, _ in os.walk(R):
        for d in list(dirs):
            if d == "__pycache__":
                shutil.rmtree(os.path.join(root, d), ignore_errors=True)
                dirs.remove(d)


ENV = dict(os.environ, PYTHONDONTWRITEBYTECODE="1")

ok = True
for name, path, old, new, test in CASES:
    before_h, before = h(path), open(path).read()
    assert old in before, f"anchor not found for: {name}"
    try:
        open(path, "w").write(before.replace(old, new, 1))
        assert h(path) != before_h, f"injection changed nothing: {name}"
        purge_bytecode()
        r = subprocess.run([f"{R}/.venv/bin/python", "-m", "pytest", test, "-q"],
                           cwd=R, capture_output=True, text=True, env=ENV)
    finally:
        # ALWAYS restore, including on an exception. An earlier version restored
        # only on the success path, and one crash left an injected defect sitting
        # in a published module until git status caught it.
        open(path, "w").write(before)
        purge_bytecode()
        assert h(path) == before_h, f"RESTORE FAILED for {path}"
    verdict = "REJECTED" if r.returncode != 0 else "*** NOT REJECTED ***"
    ok &= r.returncode != 0
    print(f"{verdict:22} {name}\n{'':22} -> {test.split('::')[-1]} exit {r.returncode}")

print("\nbaseline (no defect):")
purge_bytecode()
r = subprocess.run([f"{R}/.venv/bin/python", "-m", "pytest", T, "-q"],
                   cwd=R, capture_output=True, text=True, env=ENV)
print("   ", r.stdout.strip().splitlines()[-1], "exit", r.returncode)
ok &= r.returncode == 0
print("\nALL CONTROLS BIND" if ok else "\nA CONTROL FAILED TO BIND")
sys.exit(0 if ok else 1)
