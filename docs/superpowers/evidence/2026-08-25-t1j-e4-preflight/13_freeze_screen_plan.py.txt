#!/usr/bin/env python3
"""FREEZE the endpoint-screen plan. NOTHING HERE IS EXECUTED.

No game is played, no seed is consumed, no reference model is loaded. This
writes a plan and asserts its internal consistency -- above all that the seed
block is disjoint from every interval any earlier line reserved or consumed.
"""
import json, sys
sys.path.insert(0, "/Users/bill/projects/TwixT_Game")
from scripts.GPU.alphazero.game.twixt_state import TwixtState

# Every interval any prior line reserved or consumed, half-open [start, end).
PRIOR_INTERVALS = [
    (202608060, 202608124), (202608124, 202608188), (202608188, 202608988),
    (202608988, 202609388), (202609388, 202609788), (202609788, 202610188),
    (202611000, 202611400),          # twixtbot G3 declared block
    (202611128, 202611256),          # the re-frozen G3 schedule inside it
]
SEED_BLOCK = (202612000, 202612512)          # NEW, UNSPENT, reserved by this plan

for lo, hi in PRIOR_INTERVALS:
    assert not (SEED_BLOCK[0] < hi and lo < SEED_BLOCK[1]), \
        f"seed block {SEED_BLOCK} overlaps prior interval {(lo, hi)}"

# Balanced openings. SIX PLIES IS A HARD REQUIREMENT, not a preference: T1j's
# InitialMoves.firstMove() uses an UNSEEDED Random, and moveNr >= 6 is what keeps
# that path unreached. Every opening is verified legal and non-terminal here.
OPENINGS = {
    "o1_center":      [(11, 11), (12, 13), (13, 12), (10, 13), (12, 10), (14, 14)],
    "o2_offcenter":   [(10, 12), (13, 10), (12, 13), (11, 9), (14, 12), (9, 11)],
    "o3_low":         [(15, 11), (12, 12), (13, 10), (14, 14), (11, 11), (16, 13)],
    "o4_high":        [(8, 12), (11, 11), (10, 13), (9, 9), (12, 12), (13, 14)],
    "o5_wide_left":   [(11, 7), (12, 10), (13, 8), (10, 11), (15, 9), (14, 12)],
    "o6_wide_right":  [(11, 16), (12, 13), (13, 15), (10, 12), (15, 14), (14, 11)],
    "o7_diagonal":    [(9, 9), (12, 12), (11, 10), (14, 13), (13, 11), (16, 14)],
    "o8_contact":     [(12, 11), (11, 12), (13, 13), (12, 14), (14, 12), (10, 10)],
}
for name, mv in OPENINGS.items():
    s = TwixtState(active_size=24, to_move="red")
    for m in mv:
        assert tuple(m) in set(s.legal_moves()), f"{name}: illegal {m}"
        s = s.apply_move(tuple(m))
    assert len(mv) >= 6, f"{name}: {len(mv)} plies, need >= 6"
    assert s.winner() is None and not s.is_terminal(), f"{name}: already decided"

COLOUR_ARMS = ["t1j_red", "t1j_black"]        # balanced: every opening, both arms

tasks = []
for i, opening in enumerate(sorted(OPENINGS)):
    for j, arm in enumerate(COLOUR_ARMS):
        idx = i * len(COLOUR_ARMS) + j
        tasks.append({
            "task_id": f"e4screen-{idx:03d}-{opening}-{arm}",
            "opening": opening, "colour_arm": arm,
            "seed": SEED_BLOCK[0] + idx,
        })
assert len({t["seed"] for t in tasks}) == len(tasks), "seeds must be unique"
assert len({t["task_id"] for t in tasks}) == len(tasks), "task ids must be unique"
assert max(t["seed"] for t in tasks) < SEED_BLOCK[1], "task seeds must fit the block"
assert sum(1 for t in tasks if t["colour_arm"] == "t1j_red") == \
       sum(1 for t in tasks if t["colour_arm"] == "t1j_black"), "arms must be balanced"

plan = {
    "schema": "e4-endpoint-screen-plan/1",
    "status": "FROZEN, NOT EXECUTED — no game has been played and no seed consumed",
    "authorization": "the endpoint screen is UNAUTHORIZED; this plan exists so that "
                     "authorizing it later cannot be an occasion to choose parameters",

    "engine_endpoints": {
        "weak_mdPly": 3,
        "strong_mdPly": "READ FROM THE SWEEP'S selection RECORD — deliberately not "
                        "restated here, so the plan cannot drift from the cost rule",
        "mdFixedPly": True,
        "wall_clock_mode": "EXCLUDED — not reproducible",
    },

    "openings": {name: [list(m) for m in mv] for name, mv in OPENINGS.items()},
    "opening_plies": 6,
    "opening_plies_reason": "moveNr >= 6 keeps T1j's UNSEEDED InitialMoves.firstMove() "
                            "path unreached; this is a correctness requirement",
    "colour_arms": COLOUR_ARMS,
    "balanced": "every opening is played in both arms; 8 openings x 2 arms = 16 tasks",
    "tasks": tasks,
    "n_tasks": len(tasks),

    "seed_block": {
        "interval": list(SEED_BLOCK),
        "status": "RESERVED, UNSPENT",
        "half_open": True,
        "disjointness": "asserted against every prior interval at freeze time",
        "prior_intervals": [list(i) for i in PRIOR_INTERVALS],
        "note": "seeds are per-task and independent of T1j, whose opening randomness is "
                "bypassed rather than seeded; they exist so OUR side and task ordering "
                "are reproducible",
    },

    "ply_cap": 280,
    "per_ply_binding": "every ply of every game binds both engines through the "
                       "E3b-qualified adapter; ABORT ON THE FIRST DIVERGENCE",

    "abort_rules": [
        "any per-ply state divergence between the engines",
        "any T1j query that does not complete its requested depth",
        "any illegal or null-sentinel move from either side",
        "any postcondition failure: a Window/Frame, a non-headless jvm, a mutated "
        "host preference store, or an unauthorized reflective access",
        "any seed outside the reserved block, or any repeat of a seed",
    ],

    "durable_result_format": {
        "file": "e4_screen_results.jsonl",
        "mode": "append-only, line-buffered, never rewritten",
        "records": ["run_header", "task_start", "ply", "task_result", "stop_decision"],
        "task_result_fields": ["task_id", "opening", "colour_arm", "seed", "winner",
                               "plies", "terminal_reason", "wall_s"],
    },

    "stopping_rules": {
        "band": [0.05, 0.95],
        "band_meaning": "T1j's score rate against our side. Outside this band the screen "
                        "cannot calibrate anything, which is exactly how the twixtbot "
                        "anchor failed: it won 1.000 at every setting including no search.",
        "early_stop_check_every": 4,
        "early_stop_saturated": "stop as soon as the outcome is arithmetically outside the "
                                "band whatever the remaining tasks do",
        "outcomes": {
            "IN_BAND": "score rate inside the band; a larger match may then be proposed",
            "SATURATED_T1J_TOO_STRONG": "stop; T1j cannot calibrate against us",
            "SATURATED_T1J_TOO_WEAK": "stop; T1j cannot calibrate against us",
            "INCONCLUSIVE": "neither in-band nor saturated with the tasks run; a larger "
                            "match may then be proposed",
        },
        "explicit_inconclusive": "INCONCLUSIVE is a first-class outcome. The screen must "
                                 "never force a decision it did not earn.",
        "larger_match": "permitted ONLY if the screen lands IN_BAND or is genuinely "
                        "INCONCLUSIVE; never after a SATURATED outcome",
    },

    "claim_discipline": {
        "forbidden": [
            "strength monotonicity inferred from move differences or timing",
            "absolute strength placement — T1j is uncalibrated, so any result is an "
            "ORDERING, not a placement",
            "gating the anchor on reproducing our own ordering, which is circular",
        ],
    },
}
json.dump(plan, open(sys.argv[1], "w"), indent=1)
print(f"FROZEN, NOT EXECUTED: {len(tasks)} tasks, {len(OPENINGS)} openings x {len(COLOUR_ARMS)} arms")
print(f"seed block {SEED_BLOCK} RESERVED and UNSPENT; disjoint from {len(PRIOR_INTERVALS)} prior intervals")
print(f"seeds used by the plan: {tasks[0]['seed']}..{tasks[-1]['seed']} (block ends {SEED_BLOCK[1]})")
