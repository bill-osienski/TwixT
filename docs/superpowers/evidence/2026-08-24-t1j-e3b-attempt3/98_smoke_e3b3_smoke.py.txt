#!/usr/bin/env python3
"""E3b attempt 3 move smoke, on the COMMITTED adapter. One position, ply 3, one move.
The move is validated in OUR engine before being applied anywhere."""
import re, subprocess, sys
sys.path.insert(0, "/Users/bill/projects/TwixT_Game")
from scripts.GPU.alphazero.game.twixt_state import TwixtState
from scripts.GPU.alphazero import t1j_adapter as A

java, jar, classes = sys.argv[1:4]
CAP = 280
T1J_SEQ = [(10,10),(11,13),(12,12),(13,11),(14,10),(9,12)]
OUR_SEQ = [A.to_ours(x, y) for (x, y) in T1J_SEQ]
fails = []
def req(c, w):
    print(("  PASS   " if c else "  FAIL   ") + w)
    if not c: fails.append(w)

p = subprocess.run([java, "-Djava.util.prefs.PreferencesFactory=e2probe.ScratchPrefsFactory",
                    "-Djava.awt.headless=true", "-cp", f"{jar}:{classes}",
                    "net.schwagereit.t1j.E3bDump", "smoke"], capture_output=True, text=True)
m = re.search(r"SMOKE move_x=(-?\d+) move_y=(-?\d+) usealphabeta=(\w+) currentMaxPly=(\d+) legal=(\w+)", p.stdout)
req(p.returncode == 0, "T1j smoke exit 0")
req(m is not None, "SMOKE line emitted")
if not m: print(p.stdout); sys.exit(3)
mx, my = int(m.group(1)), int(m.group(2))
print(f"  T1j returned (x={mx}, y={my}) usealphabeta={m.group(3)} currentMaxPly={m.group(4)}")
req((mx, my) != (-1, -1), "not the null sentinel")
req(m.group(3) == "true" and int(m.group(4)) == 4, "alpha-beta completed fixed ply 3")

print("\n== validate in OUR engine BEFORE applying anywhere ==")
s = TwixtState(active_size=24, to_move="red")
for mv in OUR_SEQ: s = s.apply_move(mv)
our_move = A.to_ours(mx, my)
print(f"  T1j (x={mx}, y={my}) -> ours (row={our_move[0]}, col={our_move[1]})")
req(s.to_move == "red", "our side to move red, matching T1j Y")
req(our_move in set(s.legal_moves()), "OUR engine accepts the move as legal")
if fails: print("\nSMOKE FAIL"); sys.exit(3)

print("\n== apply to BOTH sides, compare resulting state ==")
s2 = s.apply_move(our_move)
plies, rc, _ = A.replay(OUR_SEQ + [our_move], ply_cap=CAP, java=java, jar=jar, classes=classes)
req(rc == 0, "T1j replay-with-move exit 0")
last = plies[-1]
opegs, obr = A.our_snapshot(s2)
req(opegs == last.pegs, f"pegs agree ({len(opegs)})")
req(obr == last.bridges, f"bridges agree ({len(obr)})")
req(last.next_player == A.PLAYER_TO_T1J[s2.to_move], "side to move agrees")
req(last.ply == s2.ply, f"ply agrees ({s2.ply}), each derived independently")
req(list(last.history) == [A.to_t1j(*mv) for mv in OUR_SEQ + [our_move]], "T1j ordered history matches submitted sequence")
req({A.to_t1j(r,c) for (r,c) in s2.legal_moves()} == last.legal, f"legal sets agree ({len(last.legal)})")
req(not last.term_y and not last.term_x and s2.winner() is None, "neither engine terminal")
req(A.terminal_with_cap(last.ply, last.term_y or last.term_x, ply_cap=CAP)
    == A.terminal_with_cap(s2.ply, s2.winner() is not None, ply_cap=CAP), "cap-aware terminality agrees")
print("\n" + ("SMOKE PASS" if not fails else "SMOKE FAIL"))
sys.exit(0 if not fails else 3)
