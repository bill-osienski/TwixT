"""E3b attempt 3 fixtures (unchanged from attempt 2). Each win sequence ENDS ON THE WINNING MOVE:
no earlier winner, expected winner exactly on the final ply."""
import json, sys
sys.path.insert(0, "/Users/bill/projects/TwixT_Game")
from scripts.GPU.alphazero.game.twixt_state import TwixtState

def states(moves):
    s = TwixtState(active_size=24, to_move="red")
    out = [s]
    for mv in moves:
        s = s.apply_move(tuple(mv)); out.append(s)
    return out

def alternate(primary, filler):
    out, i, j = [], 0, 0
    while i < len(primary) or j < len(filler):
        if i < len(primary): out.append(primary[i]); i += 1
        if j < len(filler):  out.append(filler[j]);  j += 1
    return out

def truncate_at_first_win(moves, expect):
    """Cut the sequence at the first ply where `expect` wins; assert none earlier."""
    st = states(moves)
    first = None
    for i, s in enumerate(st):
        w = s.winner()
        if w is not None:
            assert w == expect, f"unexpected winner {w} at ply {i}"
            first = i; break
    assert first is not None, f"{expect} never wins in this sequence"
    cut = moves[:first]
    st2 = states(cut)
    for i, s in enumerate(st2[:-1]):
        assert s.winner() is None, f"winner already at ply {i} before the final move"
    assert st2[-1].winner() == expect, "final ply must be the winning ply"
    return cut, first

red_chain, col = [], 10
for r in range(0, 23, 2):
    red_chain.append((r, col)); col += 1 if (r // 2) % 2 == 0 else -1
red_chain.append((23, red_chain[-1][1] + 2))
black_filler = [(r, c) for r in range(2, 22, 2) for c in (1, 3)][:len(red_chain)]

blk_chain, row = [], 11
for c in range(0, 23, 2):
    blk_chain.append((row, c)); row += 1 if (c // 2) % 2 == 0 else -1
blk_chain.append((blk_chain[-1][0] + 2, 23))
red_filler = [(r, c) for r in (1, 3) for c in range(2, 22, 2)][:len(blk_chain) + 1]

fixtures, report = {}, {}
fixtures["crossing"] = [(5,5),(20,20),(7,6),(20,18),(5,6),(18,20),(7,5)]
rw, rw_ply = truncate_at_first_win(alternate(red_chain, black_filler), "red")
bw, bw_ply = truncate_at_first_win(alternate(red_filler, blk_chain), "black")
fixtures["red_win"], fixtures["black_win"] = rw, bw

for name, mvs in fixtures.items():
    st = states(mvs); s = st[-1]
    earlier = [i for i, x in enumerate(st[:-1]) if x.winner() is not None]
    report[name] = {"moves": [list(m) for m in mvs], "plies": len(mvs),
                    "winner": s.winner(), "terminal": s.is_terminal(),
                    "pegs": len(s.pegs), "bridges": len(s.bridges),
                    "winner_only_on_final_ply": earlier == []}
    print(f"{name:11s} plies={len(mvs):3d} winner={s.winner()!s:6s} "
          f"pegs={len(s.pegs):3d} bridges={len(s.bridges):3d} "
          f"earlier_winner_plies={earlier}")

s = states(fixtures["crossing"])[-1]
have = {tuple(sorted(b)) for b in s.bridges}
blocked = tuple(sorted(((5,6),(7,5))))
report["crossing"]["blocked_pair"] = [list(blocked[0]), list(blocked[1])]
report["crossing"]["blocked_absent"] = blocked not in have
print(f"\ncrossing: blocked pair {blocked} absent? {blocked not in have}   formed: {sorted(have)}")
print(f"red first wins at ply {rw_ply} (fixture now {len(rw)} moves) ; "
      f"black first wins at ply {bw_ply} (fixture now {len(bw)} moves)")
json.dump(report, open(sys.argv[1], "w"), indent=1)
