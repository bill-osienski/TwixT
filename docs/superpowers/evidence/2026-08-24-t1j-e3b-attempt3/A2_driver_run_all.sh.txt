#!/bin/bash
set -o pipefail                     # attempt 2 lacked this: a piped gate reported the WRONG status
SPD=/private/tmp/claude-501/-Users-bill-projects-TwixT-Game/d037040d-d572-424f-a870-eef66233d641/scratchpad
E1=$SPD/e1; E2=$SPD/e2; E8=$SPD/e3b3
EV="$E8/evidence8"; JH="$E2/jdk/home/jdk-17.0.20.1+1/Contents/Home"
JAR="$E1/acq/release/t1j.jar"; REPO=/Users/bill/projects/TwixT_Game
JSRC="$REPO/scripts/GPU/alphazero/t1j_java"
TMPERR=$(mktemp); GATE=0
run() { printf '$ %s\n' "$1"; out=$(eval "$1" 2>"$TMPERR"); rc=$?
  if [ -n "$out" ]; then printf '%s\n' "$out" | sed 's/^/    | /'; else echo "    | <no stdout>"; fi
  if [ -s "$TMPERR" ]; then sed 's/^/    ! /' "$TMPERR"; fi
  printf '    [exit %d]\n\n' "$rc"; return $rc; }
die() { echo "*** $1 — STOPPING ***"; exit 9; }
expect() { if [ "$2" = "$3" ]; then echo "    MATCH  $1"; else echo "    *** MISMATCH $1 ***"; die "$1"; fi; }

echo "E3b ATTEMPT 3 — ADAPTER QUALIFICATION FROM COMMITTED JAVA SOURCES."
echo "Substantive commands captured as they ran."
echo "date $(date -u +%Y-%m-%dT%H:%M:%SZ)   host $(uname -srm)"; echo

echo "########## 1. IDENTITY GATE (compared) ##########"
run "git -C $E1/acq/t1j rev-parse HEAD"; expect "clone" "$out" "b572ed21f1f6b08491f69db32f531fc4b5f50fcd"
run "git -C $E1/acq/t1j status --porcelain"; [ -z "$out" ] || die "clone dirty"
run "shasum -a 256 $JAR | cut -d' ' -f1"; expect "jar" "$out" "53ec95e421db2531758142e9ee8ae49030f5345f5dc0c57b2ddb103fbd44e9b7"
run "shasum -a 256 $E2/jdk/jdk.tar.gz | cut -d' ' -f1"; expect "jdk archive" "$out" "196d13ba5f10414bef7f6a05a9b3f00edacb18ebacef2b99485db9e2ee18f0e8"
run "shasum -a 256 $JH/bin/java | cut -d' ' -f1"; expect "jvm bin/java" "$out" "af8b122943345320b179c75c3404d56a981017739746b75f9caf583632f0bea0"
run "shasum -a 256 $JH/lib/modules | cut -d' ' -f1"; expect "jvm lib/modules" "$out" "28745573641057e822a972f223fc8e40db5c9df11ae3f7402764780afc2f1951"
echo "IDENTITY GATE PASSED"; echo

echo "########## 2. ATTEMPTS 1 AND 2 AND ALL PRIOR EVIDENCE MUST BE BYTE-UNCHANGED ##########"
run "git -C $REPO status --porcelain docs/superpowers/evidence/2026-08-24-t1j-e1 docs/superpowers/evidence/2026-08-24-t1j-e2 docs/superpowers/evidence/2026-08-24-t1j-e2-attempt2 docs/superpowers/evidence/2026-08-24-t1j-e2-attempt3 docs/superpowers/evidence/2026-08-24-t1j-e2-attempt4 docs/superpowers/evidence/2026-08-24-t1j-e3a docs/superpowers/evidence/2026-08-24-t1j-e3b docs/superpowers/evidence/2026-08-24-t1j-e3b-attempt2 docs/superpowers/2026-08-24-t1j-e3b-adapter-qualification.md docs/superpowers/2026-08-24-t1j-e3b-attempt2.md"
[ -z "$out" ] || die "prior attempts or evidence dirty"; echo

echo "########## 3. PREFS BEFORE ##########"
run "bash $E2/prefsfp.sh BEFORE | tee $EV/91_prefs_before.txt"; echo

echo "########## 4. MANIFEST OF EVERY SOURCE AND COMPILE INPUT ##########"
echo "    attempt 2 hashed 7 files and omitted BOTH Preferences compile inputs."
run "shasum -a 256 $REPO/scripts/GPU/alphazero/t1j_adapter.py $REPO/tests/test_t1j_adapter.py $JSRC/net/schwagereit/t1j/E3bDump.java $JSRC/e2probe/ScratchPrefs.java $JSRC/e2probe/ScratchPrefsFactory.java $E8/e3b3_compare.py $E8/e3b3_smoke.py $E8/build_fixtures.py $E8/check_prefs.py $E8/compile_via_adapter.py $E8/run_all.sh"
echo

echo "########## 5. ADAPTER UNIT TESTS — REAL EXIT STATUS, GATED ##########"
run "$REPO/.venv/bin/python -m pytest $REPO/tests/test_t1j_adapter.py -q --no-header -p no:cacheprovider"
TESTS=$rc; echo "    unit-test exit: $TESTS (no pipe: this IS pytest's status)"; echo

echo "########## 6. COMPILE THROUGH THE COMMITTED ADAPTER, FROM COMMITTED SOURCES ##########"
run "rm -rf $E8/classes && python3 $E8/compile_via_adapter.py $JH/bin/javac $JAR $E8/classes; echo \$?"
[ "$(echo "$out" | tail -1)" = "0" ] || die "compile through the adapter failed"
run "find $E8/classes -name '*.class' | sort | xargs shasum -a 256"; echo

echo "########## 7. LEGALITY DUMP + FIXTURES (fixtures end ON the winning move) ##########"
run "$JH/bin/java -Djava.util.prefs.PreferencesFactory=e2probe.ScratchPrefsFactory -Djava.awt.headless=true -cp $JAR:$E8/classes net.schwagereit.t1j.E3bDump legality > $EV/92_legality.txt; echo \$?"

SMOKE=$out; run "cat $EV/95_smoke.txt"; echo

echo "########## 10. PREFS CHECKER — REJECTION CONTROLS FIRST, THEN THE REAL PAIR ##########"
run "bash $E2/prefsfp.sh AFTER | tee $EV/96_prefs_after.txt"
echo "    controls: the checker must REJECT a mutated fingerprint before its PASS means anything."
PREFSCTRL=0
run "python3 - $EV/91_prefs_before.txt $EV/96_prefs_after.txt $EV/A4_prefs_control.txt <<'PYCTL'
import subprocess, sys
before, after, outfile = sys.argv[1:4]
lines = open(after).read().splitlines(True)
def field_idx():
    for i, l in enumerate(lines):
        if ':' in l and not l.startswith('host preference store fingerprint'):
            return i
    raise SystemExit('no normalized field to mutate')
i = field_idx()
k, v = lines[i].split(':', 1)
log, bad = [], 0
def case(name, mutated_text, want):
    global bad
    open('/tmp/.e3b3_prefs_ctl', 'w').write(mutated_text)
    r = subprocess.run([sys.executable, '$E8/check_prefs.py', before, '/tmp/.e3b3_prefs_ctl'],
                       capture_output=True, text=True)
    ok = r.returncode == want
    log.append(f\"  {'PASS' if ok else 'FAIL'}  {name}: exit {r.returncode} (want {want})\")
    log.append('        ' + r.stdout.strip().replace(chr(10), ' | '))
    if not ok: bad += 1
mut = list(lines); mut[i] = f'{k}: MUTATED-{v.strip()}\n'
case('mutated field value rejected', ''.join(mut), 3)
drop = [l for j, l in enumerate(lines) if j != i]
case('dropped field rejected', ''.join(drop), 3)
case('untouched AFTER accepted (the control control)', ''.join(lines), 0)
open(outfile, 'w').write(chr(10).join(log) + chr(10))
print(chr(10).join(log))
print('PREFS CONTROLS ' + ('OK' if not bad else f'FAILED ({bad})'))
sys.exit(0 if not bad else 3)
PYCTL"
PREFSCTRL=$rc
run "python3 $E8/check_prefs.py $EV/91_prefs_before.txt $EV/96_prefs_after.txt; echo \$?"
PREFS=$(echo "$out" | tail -1)
echo "    prefs controls exit: $PREFSCTRL   prefs real-pair exit: $PREFS"; echo
run "ls /Library/Java/JavaVirtualMachines | wc -l"
rm -f "$TMPERR" /tmp/.e3b3_prefs_ctl
echo "TRANSCRIPT COMPLETE — qualification $QUAL, smoke $SMOKE, unit tests $TESTS, prefs-controls $PREFSCTRL, prefs $PREFS"
[ "$QUAL" = "0" ] && [ "$SMOKE" = "0" ] && [ "$TESTS" = "0" ] \
  && [ "$PREFSCTRL" = "0" ] && [ "$PREFS" = "0" ] || GATE=3
echo "GATE $GATE"
exit $GATE
