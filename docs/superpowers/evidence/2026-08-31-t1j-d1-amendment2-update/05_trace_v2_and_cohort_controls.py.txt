def purge():
    for d in pathlib.Path(".").rglob("__pycache__"):
        shutil.rmtree(d, ignore_errors=True)


FILES = sorted({f for _l, f, _o, _n, _t in DEFECTS})
ORIGINAL = {f: pathlib.Path(f).read_text() for f in FILES}

ok = stale = missed = 0
try:
    for label, path, old, new, test in DEFECTS:
        src = ORIGINAL[path]
        n = src.count(old)
        if n != 1:
            print(f"  STALE CONTROL -- DID NOT RUN  {label}: anchor matched {n} times "
                  f"in {path}")
            stale += 1
            continue
        pathlib.Path(path).write_text(src.replace(old, new))
        purge()
        r = subprocess.run(
            [sys.executable, "-m", "pytest", test, "-q", "-p", "no:cacheprovider"],
            capture_output=True, text=True,
            env={**os.environ, "PYTHONDONTWRITEBYTECODE": "1"})
        pathlib.Path(path).write_text(src)
        rejected = r.returncode != 0
        print(f"  {'REJECTED' if rejected else 'NOT CAUGHT':11s}  {label}")
        if rejected:
            ok += 1
        else:
            missed += 1
            print("      " + r.stdout.strip().splitlines()[-1][:160])
finally:
    for f in FILES:
        pathlib.Path(f).write_text(ORIGINAL[f])
    purge()

print(f"\n{ok}/{len(DEFECTS)} defects rejected; {missed} not caught; {stale} stale")
print("ALL CONTROLS RAN:", stale == 0)
print("every source restored:",
      all(pathlib.Path(f).read_text() == ORIGINAL[f] for f in FILES))
