#!/bin/bash
E1=/private/tmp/claude-501/-Users-bill-projects-TwixT-Game/d037040d-d572-424f-a870-eef66233d641/scratchpad/e1
E2=/private/tmp/claude-501/-Users-bill-projects-TwixT-Game/d037040d-d572-424f-a870-eef66233d641/scratchpad/e2
E6=/private/tmp/claude-501/-Users-bill-projects-TwixT-Game/d037040d-d572-424f-a870-eef66233d641/scratchpad/e3b
EV="$E6/evidence6"; JH="$E2/jdk/home/jdk-17.0.20.1+1/Contents/Home"
JAR="$E1/acq/release/t1j.jar"; REPO=/Users/bill/projects/TwixT_Game
TMPERR=$(mktemp)
run() { printf '$ %s\n' "$1"; out=$(eval "$1" 2>"$TMPERR"); rc=$?
  if [ -n "$out" ]; then printf '%s\n' "$out" | sed 's/^/    | /'; else echo "    | <no stdout>"; fi
  if [ -s "$TMPERR" ]; then sed 's/^/    ! /' "$TMPERR"; fi
  printf '    [exit %d]\n\n' "$rc"; return $rc; }
die() { echo "*** $1 — STOPPING ***"; exit 9; }
expect() { if [ "$2" = "$3" ]; then echo "    MATCH  $1"; else echo "    *** MISMATCH $1 ***"; die "$1"; fi; }

echo "E3b — ADAPTER QUALIFICATION. Substantive commands captured as they ran."
echo "date $(date -u +%Y-%m-%dT%H:%M:%SZ)   host $(uname -srm)"
echo

echo "########## 1. IDENTITY GATE (compared, not recorded) ##########"
run "git -C $E1/acq/t1j rev-parse HEAD"; expect "clone" "$out" "b572ed21f1f6b08491f69db32f531fc4b5f50fcd"
run "git -C $E1/acq/t1j status --porcelain"; [ -z "$out" ] || die "clone dirty"
run "shasum -a 256 $JAR | cut -d' ' -f1"; expect "jar" "$out" "53ec95e421db2531758142e9ee8ae49030f5345f5dc0c57b2ddb103fbd44e9b7"
run "shasum -a 256 $E2/jdk/jdk.tar.gz | cut -d' ' -f1"; expect "jdk archive" "$out" "196d13ba5f10414bef7f6a05a9b3f00edacb18ebacef2b99485db9e2ee18f0e8"
run "shasum -a 256 $JH/bin/java | cut -d' ' -f1"; expect "jvm bin/java" "$out" "af8b122943345320b179c75c3404d56a981017739746b75f9caf583632f0bea0"
run "shasum -a 256 $JH/lib/modules | cut -d' ' -f1"; expect "jvm lib/modules" "$out" "28745573641057e822a972f223fc8e40db5c9df11ae3f7402764780afc2f1951"
echo "IDENTITY GATE PASSED"; echo

echo "########## 2. ALL PRIOR E1/E2/E3a EVIDENCE MUST BE BYTE-UNCHANGED ##########"
run "git -C $REPO status --porcelain docs/superpowers/evidence/2026-08-24-t1j-e1 docs/superpowers/evidence/2026-08-24-t1j-e2 docs/superpowers/evidence/2026-08-24-t1j-e2-attempt2 docs/superpowers/evidence/2026-08-24-t1j-e2-attempt3 docs/superpowers/evidence/2026-08-24-t1j-e2-attempt4 docs/superpowers/evidence/2026-08-24-t1j-e3a"
[ -z "$out" ] || die "prior evidence dirty"; echo

echo "########## 3. PREFS BEFORE ##########"
run "bash $E2/prefsfp.sh BEFORE | tee $EV/71_prefs_before.txt"; echo

echo "########## 4. MANIFEST: adapter + helper sources ##########"
run "shasum -a 256 $E6/src/net/schwagereit/t1j/E3bDump.java $E6/e3b_compare.py $E6/e3b_smoke.py $E6/build_fixtures.py"
echo

echo "########## 5. COMPILE (helper only; T1j untouched) ##########"
run "$JH/bin/javac -Xlint:-options -encoding UTF-8 -cp $JAR -d $E6/classes $E6/src/e2probe/ScratchPrefs.java $E6/src/e2probe/ScratchPrefsFactory.java $E6/src/net/schwagereit/t1j/E3bDump.java"
[ $rc -eq 0 ] || die "compile failed"
run "find $E6/classes -name '*.class' | sort | xargs shasum -a 256"; echo

echo "########## 6. T1j LEGALITY DUMP (empty board, both players) ##########"
run "$JH/bin/java -Djava.util.prefs.PreferencesFactory=e2probe.ScratchPrefsFactory -Djava.awt.headless=true -cp $JAR:$E6/classes net.schwagereit.t1j.E3bDump legality > $EV/72_legality.txt; echo \$?"
[ "$out" = "0" ] || die "legality dump failed"
run "head -3 $EV/72_legality.txt; tail -1 $EV/72_legality.txt"; echo

echo "########## 7. FIXTURES, BUILT AND VERIFIED AGAINST OUR ENGINE ##########"
run "python3 $E6/build_fixtures.py $EV/73_fixtures.json"; echo

echo "########## 8. ADAPTER + LOCKSTEP COMPARATOR (self-test, derivation, qualification) ##########"
run "python3 $E6/e3b_compare.py $JH/bin/java $JAR $E6/classes $EV/72_legality.txt $EV/73_fixtures.json > $EV/74_qualification.txt 2>&1; echo \$?"
QUAL=$out; run "cat $EV/74_qualification.txt"
echo

echo "########## 9. MOVE SMOKE (one fixed position, ply 3, one move) ##########"
run "python3 $E6/e3b_smoke.py $JH/bin/java $JAR $E6/classes $E6 > $EV/75_smoke.txt 2>&1; echo \$?"
SMOKE=$out; run "cat $EV/75_smoke.txt"
echo

echo "########## 10. PREFS AFTER + SYSTEM ##########"
run "bash $E2/prefsfp.sh AFTER | tee $EV/76_prefs_after.txt"
run "diff $EV/71_prefs_before.txt $EV/76_prefs_after.txt"
run "ls /Library/Java/JavaVirtualMachines | wc -l"
rm -f "$TMPERR"
echo "TRANSCRIPT COMPLETE — qualification exit $QUAL, smoke exit $SMOKE"
[ "$QUAL" = "0" ] && [ "$SMOKE" = "0" ] || exit 3
exit 0
