#!/usr/bin/env python3
"""E3b move smoke: one fixed position, fixed ply 3, one computeMove.

The returned move is VALIDATED IN OUR ENGINE FIRST. Only then is it applied to
both sides and the resulting states compared. Not a game: one move.
"""
import re, subprocess, sys
sys.path.insert(0, "/Users/bill/projects/TwixT_Game")
sys.path.insert(0, sys.argv[4])
from scripts.GPU.alphazero.game.twixt_state import TwixtState
from e3b_compare import TRANSFORMS, PLAYER_MAPS, our_snapshot, parse_plies, t1j_replay

java, jar, classes = sys.argv[1:4]
tf, pm = TRANSFORMS["identity"], PLAYER_MAPS["Y=red"]
inv = lambda x, y: (y, x)          # T1j (x,y) -> our (row,col), inverse of identity

T1J_SEQ = [(10,10),(11,13),(12,12),(13,11),(14,10),(9,12)]   # T1j coords, as in E2/E3a
OUR_SEQ = [inv(x, y) for (x, y) in T1J_SEQ]

fails = []
def req(c, what):
    print(("  PASS   " if c else "  FAIL   ") + what)
    if not c: fails.append(what)

print("== smoke: T1j computes one move at fixed ply 3 ==")
p = subprocess.run([java, "-Djava.util.prefs.PreferencesFactory=e2probe.ScratchPrefsFactory",
                    "-Djava.awt.headless=true", "-cp", f"{jar}:{classes}",
                    "net.schwagereit.t1j.E3bDump", "smoke"], capture_output=True, text=True)
print("  T1j exit", p.returncode)
m = re.search(r"SMOKE move_x=(-?\d+) move_y=(-?\d+) usealphabeta=(\w+) currentMaxPly=(\d+) legal=(\w+)",
              p.stdout)
req(p.returncode == 0, "T1j smoke process exit 0")
req(m is not None, "SMOKE line emitted")
if not m: print(p.stdout); sys.exit(3)
mx, my, uab, cmp_, t1j_legal = int(m.group(1)), int(m.group(2)), m.group(3), int(m.group(4)), m.group(5)
print(f"  T1j returned (x={mx}, y={my}) usealphabeta={uab} currentMaxPly={cmp_}")
req((mx, my) != (-1, -1), "returned move is not the null sentinel")
req(uab == "true" and cmp_ == 4, "alpha-beta completed fixed ply 3")
req(t1j_legal == "true", "T1j considers its own move legal")

print()
print("== validate the returned move IN OUR ENGINE, before applying it anywhere ==")
s = TwixtState(active_size=24, to_move="red")
for mv in OUR_SEQ:
    s = s.apply_move(mv)
our_move = inv(mx, my)
print(f"  T1j (x={mx}, y={my})  ->  ours (row={our_move[0]}, col={our_move[1]})")
req(s.to_move == "red", "our side to move is red, matching T1j's Y")
req(our_move in set(s.legal_moves()), "OUR engine accepts the move as legal")
req(s.pegs.get(our_move) is None, "target intersection empty in our engine")
if fails:
    print("\nSMOKE FAIL"); sys.exit(3)

print()
print("== apply to BOTH sides and compare the resulting state ==")
s2 = s.apply_move(our_move)
t_moves = [tf(r, c) for (r, c) in OUR_SEQ] + [(mx, my)]
out, rc = t1j_replay(java, jar, classes, t_moves, ply_cap=280)
req(rc == 0, "T1j replay-with-move exit 0")
plies = parse_plies(out)
last = plies[-1]
opegs, obr = our_snapshot(s2, tf, pm)
req(opegs == last["pegs"], f"pegs agree after the move ({len(opegs)} each)")
req(obr == last["bridges"], f"bridges agree after the move ({len(obr)} each)")
req(last["hdr"].get("next") == pm[s2.to_move], "side to move agrees after the move")
req(int(last["hdr"].get("moveNr")) == s2.ply, f"ply agrees after the move ({s2.ply})")
req(last["hdr"].get("termY") == "false" and last["hdr"].get("termX") == "false"
    and s2.winner() is None, "neither engine reports a terminal state")
print()
print("SMOKE PASS" if not fails else "SMOKE FAIL")
sys.exit(0 if not fails else 3)
