"""A pytest plugin that makes loading the incumbent checkpoint FATAL.

`checkpoints/alphazero-v2-calib020-from0409/model_iter_0001.safetensors` exists
in this tree, so a test that reached `_default_load_evaluator` would quietly
load a real model and the run would still be green. Asserting "no model was
loaded" is not the same as observing it. This turns the load into a failure, so
a green run under this plugin is the observation.

`e4_screen_command._default_load_evaluator` imports the loader INSIDE the
function, so rebinding the module attribute here is seen at call time.
"""


def pytest_configure(config):
    from scripts.GPU.alphazero import twixtbot_g3_reference as G3

    def refuse(reference, repo_root):
        raise AssertionError(
            f"a test reached load_reference_evaluator({reference!r}) -- a real "
            f"checkpoint would have been read from {repo_root!r}")

    G3.load_reference_evaluator = refuse
