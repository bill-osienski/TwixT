def purge():
    for d in pathlib.Path(".").rglob("__pycache__"):
        shutil.rmtree(d, ignore_errors=True)


FILES = sorted({f for _l, f, _o, _n, _t in DEFECTS})
ORIGINAL = {f: pathlib.Path(f).read_text() for f in FILES}

# ═══════════════════════ THE CLEAN BASELINE, FIRST ═══════════════════════════
# 🔴 The harness never checked that a control's target test PASSES on unmodified
# source. A test that is already broken FAILS under the defect too, so its
# controls reported REJECTED for free -- a control reporting success for work it
# never did, which is the exact failure this harness exists to catch. Found when
# a docstring test with a bad assertion sat under three green controls.
#
# One pytest invocation over every distinct target node, before anything is
# injected. If any fails here, no control naming it means anything.
NODES = sorted({t for *_r, t in DEFECTS})
purge()
_b = subprocess.run(
    [sys.executable, "-m", "pytest", *NODES, "-q", "-p", "no:cacheprovider"],
    capture_output=True, text=True,
    env={**os.environ, "PYTHONDONTWRITEBYTECODE": "1"})
BASELINE_OK = _b.returncode == 0
if not BASELINE_OK:
    print("🔴 BASELINE FAILURES -- these tests do not pass on CLEAN source, so every "
          "control naming them is meaningless:")
    # STDERR TOO: a missing node id is reported there, not on stdout, so the old
    # stdout-only scan printed the FAIL verdict with no reason under it.
    for line in (_b.stdout + "\n" + _b.stderr).splitlines():
        if line.startswith(("FAILED", "ERROR", "  (no match")):
            print("   ", line[:150])
print(f"clean baseline over {len(NODES)} distinct target tests: "
      f"{'PASS' if BASELINE_OK else 'FAIL'}")

ok = stale = missed = 0
try:
    for label, path, old, new, test in DEFECTS:
        src = ORIGINAL[path]
        n = src.count(old)
        if n != 1:
            print(f"  STALE CONTROL -- DID NOT RUN  {label}: anchor matched {n} times "
                  f"in {path}")
            stale += 1
            continue
        pathlib.Path(path).write_text(src.replace(old, new))
        purge()
        r = subprocess.run(
            [sys.executable, "-m", "pytest", test, "-q", "-p", "no:cacheprovider"],
            capture_output=True, text=True,
            env={**os.environ, "PYTHONDONTWRITEBYTECODE": "1"})
        pathlib.Path(path).write_text(src)
        rejected = r.returncode != 0
        print(f"  {'REJECTED' if rejected else 'NOT CAUGHT':11s}  {label}")
        if rejected:
            ok += 1
        else:
            missed += 1
            print("      " + r.stdout.strip().splitlines()[-1][:160])
finally:
    for f in FILES:
        pathlib.Path(f).write_text(ORIGINAL[f])
    purge()

print(f"\n{ok}/{len(DEFECTS)} defects rejected; {missed} not caught; {stale} stale")
print("ALL CONTROLS RAN:", stale == 0 and BASELINE_OK)
print("clean baseline:", "PASS" if BASELINE_OK else "🔴 FAIL")
print("every source restored:",
      all(pathlib.Path(f).read_text() == ORIGINAL[f] for f in FILES))
