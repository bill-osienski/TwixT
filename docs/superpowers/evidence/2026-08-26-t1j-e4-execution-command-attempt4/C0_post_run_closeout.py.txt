AMENDED = {

def git(*a):
    r = subprocess.run(["git", "-C", REPO, *a], capture_output=True, text=True)
    return r.returncode, r.stdout.strip()


def blob(commit, path):
    rc, out = git("rev-parse", f"{commit}:{path}")
    return out if rc == 0 else None


def files_at(commit, prefix):
    rc, out = git("ls-tree", "-r", "--format=%(objectname) %(path)", commit, "--", prefix)
    if rc != 0 or not out:
        return {}
    return {l.split(" ", 1)[1]: l.split(" ", 1)[0] for l in out.splitlines()}


def added_in(path):
    rc, out = git("log", "--diff-filter=A", "--format=%H", "-1", "--", path)
    return out.splitlines()[0] if rc == 0 and out else None


def check_rows(manifest_text, commit, blob_of, amended_here):
    log, bad = [], 0
    rows = {m.group(3): (m.group(1), m.group(2))
            for m in (ROW_RE.match(l) for l in manifest_text.splitlines()) if m}
    if not rows:
        return 1, ["FAIL manifest has no parsable rows"]
    for path in sorted(rows):
        _, want = rows[path]
        got = blob_of(commit, path)
        ok = got is not None and got == want
        note = "  (amended by the closeout; see below)" if path in amended_here else ""
        log.append(f"{'PASS' if ok else 'FAIL'}  {path}{note}")
        log.append(f"        manifest blob    {want}")
        log.append(f"        {commit}:<path>  {got}")
        if not ok:
            bad += 1
    return bad, log


if __name__ == "__main__":
    manifest = sys.argv[1]
    mf = open(manifest).read()
    print(f"E4 EXECUTION-COMMAND POST-RUN CLOSEOUT — commit {COMMIT}\n")

    print("A. attempt-4 tracked manifest rows vs the COMMIT's objects")
    bad_a, log_a = check_rows(mf, COMMIT, blob, AMENDED_BY_THIS_COMMIT)

    ok_b = obj is None

        for path, b in sorted(now.items()):

            if o is None or files_at(o, path).get(path) != b:
                changed.append(path)
            elif o:
                origins[o[:7]] = origins.get(o[:7], 0) + 1
        ok = not changed
        print(f"{'PASS' if ok else 'FAIL'}  {prefix}")
        print(f"        {len(now)} files vs their origin commits: "
              + ", ".join(f"{c}x{n}" for c, n in sorted(origins.items()))
              + f"  changed {len(changed)}")
        if not ok:
            print(f"        changed={changed[:3]}"); bad_c += 1
    print(f"C: {'OK' if not bad_c else f'FAILED ({bad_c})'}\n")

    print("D. DECLARED amendments — checked against the amending commit, and required")
    print("   to actually differ from their origin")
    bad_d = 0
    for path, (origin, amender, why) in sorted(AMENDED.items()):
        a, b, now = blob(origin, path), blob(amender, path), blob(COMMIT, path)
        ok = a and b and a != b and b == now
        print(f"  {'PASS' if ok else 'FAIL'}  {path}  ({origin} -> {amender})")
        print(f"        why: {why}")
        if not ok:
            bad_d += 1
    print()
    print("   AMENDED BY THIS CLOSEOUT -- its commit hash cannot be known while it is")
    print("   being written, so the worktree blob is recorded and must differ from the")
    print("   committed one. Verify after committing: git rev-parse <closeout>:<path>")
    for path, why in sorted(AMENDED_BY_THIS_COMMIT.items()):
        at_commit = blob(COMMIT, path)
        rc, wt = git("hash-object", path)
        ok = rc == 0 and wt and at_commit and wt != at_commit
        print(f"  {'PASS' if ok else 'FAIL'}  {path}")
        print(f"        at {COMMIT}   {at_commit}")
        print(f"        worktree     {wt}")
        print(f"        why: {why}")
        if not ok:
            bad_d += 1
    print(f"D: {'OK' if not bad_d else f'FAILED ({bad_d})'}\n")

    print("negative controls — the SAME checker, tampered:")

    rows = {m.group(3): (m.group(1), m.group(2))

    first = sorted(rows)[0]
    flip = lambda h: ("f" if h[0] != "f" else "0") + h[1:]      # noqa: E731

    print(f"CONTROLS {'OK' if not ctl else f'FAILED ({ctl})'}")
    sys.exit(0 if not (bad_a or not ok_b or bad_c or bad_d or ctl) else 3)
