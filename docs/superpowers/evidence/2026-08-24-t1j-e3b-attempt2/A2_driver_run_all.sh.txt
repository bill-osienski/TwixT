E7=/private/tmp/claude-501/-Users-bill-projects-TwixT-Game/d037040d-d572-424f-a870-eef66233d641/scratchpad/e3b2

run() { printf '$ %s\n' "$1"; out=$(eval "$1" 2>"$TMPERR"); rc=$?
  if [ -n "$out" ]; then printf '%s\n' "$out" | sed 's/^/    | /'; else echo "    | <no stdout>"; fi
  if [ -s "$TMPERR" ]; then sed 's/^/    ! /' "$TMPERR"; fi
  printf '    [exit %d]\n\n' "$rc"; return $rc; }
die() { echo "*** $1 — STOPPING ***"; exit 9; }
expect() { if [ "$2" = "$3" ]; then echo "    MATCH  $1"; else echo "    *** MISMATCH $1 ***"; die "$1"; fi; }

echo "E3b ATTEMPT 2 — ADAPTER QUALIFICATION. Substantive commands captured as they ran."

echo "########## 1. IDENTITY GATE (compared) ##########"
run "git -C $E1/acq/t1j rev-parse HEAD"; expect "clone" "$out" "b572ed21f1f6b08491f69db32f531fc4b5f50fcd"

run "shasum -a 256 $JAR | cut -d' ' -f1"; expect "jar" "$out" "53ec95e421db2531758142e9ee8ae49030f5345f5dc0c57b2ddb103fbd44e9b7"

run "shasum -a 256 $JH/lib/modules | cut -d' ' -f1"; expect "jvm lib/modules" "$out" "28745573641057e822a972f223fc8e40db5c9df11ae3f7402764780afc2f1951"

echo "########## 2. ATTEMPT 1 AND ALL PRIOR EVIDENCE MUST BE BYTE-UNCHANGED ##########"
run "git -C $REPO status --porcelain docs/superpowers/evidence/2026-08-24-t1j-e1 docs/superpowers/evidence/2026-08-24-t1j-e2 docs/superpowers/evidence/2026-08-24-t1j-e2-attempt2 docs/superpowers/evidence/2026-08-24-t1j-e2-attempt3 docs/superpowers/evidence/2026-08-24-t1j-e2-attempt4 docs/superpowers/evidence/2026-08-24-t1j-e3a docs/superpowers/evidence/2026-08-24-t1j-e3b"

run "bash $E2/prefsfp.sh BEFORE | tee $EV/91_prefs_before.txt"; echo

run "shasum -a 256 $REPO/scripts/GPU/alphazero/t1j_adapter.py $REPO/tests/test_t1j_adapter.py $E7/src/net/schwagereit/t1j/E3bDump.java $E7/e3b2_compare.py $E7/e3b2_smoke.py $E7/build_fixtures.py $E7/check_prefs.py"

run "$REPO/.venv/bin/python -m pytest $REPO/tests/test_t1j_adapter.py -q 2>&1 | tail -3"

echo "########## 6. COMPILE HELPER (T1j untouched) ##########"
